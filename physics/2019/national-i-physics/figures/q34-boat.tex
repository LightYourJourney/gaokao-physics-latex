% 第 34 题(2)：帆船桅杆与湖底 P 点发出的激光（水面处折射）
% 几何：P 在水下 4 m，出射点 S 在水面；水中段 P→S 与竖直方向成 θ，水面以上段 S→顶端与竖直方向成 53°
\begin{tikzpicture}[font=\small, >=Stealth, scale=0.86]
  % 水体
  \fill[blue!7] (-0.6,0.5) rectangle (9.6,2.75);
  % 湖底
  \draw[thick] (-0.6,0.5) -- (9.6,0.5);
  \foreach \x in {-0.45,-0.1,...,9.5} { \draw (\x,0.5) -- (\x-0.13,0.32); }
  % 水面
  \draw[thick] (-0.6,2.75) -- (9.6,2.75);
  % 桅杆
  \draw[very thick] (7.6,2.75) -- (7.6,5.55);
  \node[above,font=\scriptsize] at (7.6,5.62) {顶端};
  % 桅杆高 3 m
  \draw[<->] (8.25,2.75) -- (8.25,5.55);
  \node[right,font=\scriptsize] at (8.28,4.15) {\SI{3}{\metre}};
  % 水深 4 m
  \draw[<->] (9.05,0.5) -- (9.05,2.75);
  \node[right,font=\scriptsize] at (9.08,1.6) {\SI{4}{\metre}};
  % P 点（水下 4 m，位于出射点 S 的左侧 3 m）
  \coordinate (P) at (2.0,0.5);
  \fill (P) circle (1.8pt);
  \node[below,font=\scriptsize] at (2.0,0.38) {$P$};
  % 出射点 S（水面）与法线
  \coordinate (S) at (5.0,2.75);
  \draw[dashed] (S) -- (5.0,0.85);
  \draw[dashed] (S) -- (5.0,4.35);
  % 水中光线：P → S
  \draw[thick,->] (P) -- (S);
  % 水面以上光线：S → 桅杆顶端
  \coordinate (Top) at (7.6,5.55);
  \draw[thick,->] (S) -- (Top);
  % 53°（水面以上，与法线的夹角）
  \coordinate (Vup) at (5.0,4.35);
  \pic[draw, angle radius=0.85cm, angle eccentricity=1.32, "$53^{\circ}$"]
      {angle = Top--S--Vup};
  % θ（水中，与法线的夹角）
  \coordinate (Vdn) at (5.0,0.85);
  \pic[draw, angle radius=0.7cm, angle eccentricity=1.4, "$\theta$"]
      {angle = P--S--Vdn};
  % 水面标注
  \node[left,font=\scriptsize] at (-0.62,2.75) {水面};
\end{tikzpicture}
