% 第 2 题图乙：两谷粒的运动轨迹（TikZ 重绘）
% 谷粒 1 平抛（水平初速度 v1），谷粒 2 斜上抛（初速度 v2），两轨迹交于 P 点
\begin{tikzpicture}[font=\small, >=Stealth, scale=0.9]
  % 谷粒 1：平抛轨迹 y = -0.2 x^2
  \draw[dashed, thick, domain=0:4, samples=80, smooth]
    plot (\x, {-0.2*\x*\x});
  % 谷粒 2：斜上抛轨迹 y = -(2/3)x^2 + 1.867x，过 O(0,0) 与 P(4,-3.2)
  \draw[dashed, thick, domain=0:4, samples=80, smooth]
    plot (\x, {-0.667*\x*\x+1.867*\x});
  % 抛出点 O 与交点 P
  \fill (0,0) circle (1.6pt) node[below left]{$O$};
  \fill (4,-3.2) circle (1.6pt) node[right]{$P$};
  % 初速度方向
  \draw[->, thick] (0,0) -- (1.15,0);
  \node at (0.85,0.18) {$v_1$};
  \draw[->, thick] (0,0) -- (0.72,1.34);
  \node at (0.5,1.55) {$v_2$};
  % 轨迹标注
  \node at (1.25,-1.15) {谷粒 1};
  \node at (2.62,1.3) {谷粒 2};
  \node[below] at (2.0,-3.55) {乙};
\end{tikzpicture}
