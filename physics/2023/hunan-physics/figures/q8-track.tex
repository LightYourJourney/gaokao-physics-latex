% 第 8 题：光滑轨道 ABC（直线段 AB 与圆弧段 BC 相切于 B，TikZ 重绘）
% 作图取 R = 1.1、θ = 30°，圆心 O 在 C 正下方 R 处，A 比 C 低 2R
\begin{tikzpicture}[font=\small, >=Stealth]
  \coordinate (A) at (-2.90,0.4);
  \coordinate (B) at (0.65,2.4526);
  \coordinate (O) at (1.2,1.5);
  \coordinate (C) at (1.2,2.6);
  \coordinate (G) at (-1.6,0.4);
  % 地面
  \draw[thick] (-3.4,0.4) -- (1.55,0.4);
  \foreach \x in {-3.3,-3.0,...,1.4} {
    \draw (\x,0.4) -- (\x-0.12,0.16);}
  % 轨道：直线段 AB
  \draw[very thick] (A) -- (B);
  % 圆弧段 BC（圆心 O，半径 1.1，从 120° 到 90°）
  \draw[very thick] (O) ++(120:1.1) arc[start angle=120, end angle=90, radius=1.1];
  % 右侧竖直支护
  \draw (1.2,0.4) -- (C);
  % 圆心与半径
  \fill (O) circle (1.4pt) node[right]{$O$};
  \draw[dashed] (O) -- (B);
  % A、B、C
  \fill (A) circle (1.4pt) node[below left]{$A$};
  \fill (B) circle (2.2pt);
  \node[above left] at (B) {$B$};
  \node[above] at (C) {$C$};
  % 初速度
  \draw[->, thick] (-2.82,0.45) -- (-2.15,0.84) node[above left, pos=0.6]{$v_0$};
  % 角度
  \pic[draw, angle radius=0.5cm, angle eccentricity=1.55, "$\theta$"]
      {angle = G--A--B};
  \pic[draw, angle radius=0.6cm] {angle = C--O--B};
  \node at (0.98,1.9) {$\alpha$};
\end{tikzpicture}
