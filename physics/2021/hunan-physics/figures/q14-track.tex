% 第 14 题：倾斜轨道 + 水平轨道 + 弧形轨道 PQ 示意图
% x 轴水平向右、y 轴竖直向下（O 为原点）；P(2μl, μl)、Q 在 y 轴上；
% PQ 为椭圆弧（解析：x^2+4y^2-8μly=0，中心 (0,μl)，x 半轴 2μl、y 半轴 μl）。
\begin{tikzpicture}[font=\small, >=Stealth, scale=1.05]
  \def\muL{1.0}   % 单位长度取 μl
  % 坐标系（y 向下：tikz 中取负）
  \draw[->] (-3.2,0) -- (4.2,0) node[right] {$x$};
  \draw[->] (0,0.5) -- (0,-3.4) node[below] {$y$};
  % 倾斜轨道（左上 → 右下，与水平轨道平滑连接）
  \draw[thick] (-3.0,1.9) -- (-1.0,0.8);
  % 水平轨道（长 l = 1×μl，从连接处到 O）
  \draw[thick] (-1.0,0.8) -- (0,0.8);
  % 弧形轨道 PQ：从 P(2μl, μl) 到 Q(0, 2μl)（y 向下坐标）
  \draw[thick] plot[domain=90:180, samples=50]
      ({2*\muL*sin(\x)}, {-\muL+\muL*cos(\x)});
  \fill (0,0) circle (1.5pt) node[above left] {$O$};
  \fill ({2*\muL},{-\muL}) circle (1.5pt) node[right] {$P$};
  \fill (0,{-2*\muL}) circle (1.5pt) node[left] {$Q$};
  % 标注
  \draw[<->] (0.2,-0.4) -- (0.2,{-2*\muL+0.2}) node[midway, right] {$2\mu l$};
  \draw[<->] (-1.3,0.95) -- (-0.1,0.95) node[midway, above] {$l$};
  \draw[dashed, gray] ({2*\muL},{-\muL}) -- (0,{-\muL});
  \node[gray] at ({2*\muL+0.55},-0.9) {弧形轨道};
  \node at (-2.2,1.35) {倾斜轨道};
\end{tikzpicture}
