\begin{tikzpicture}[font=\small, >=Stealth]
  \def\a{1.6}

  \draw[->] (0,-0.6) -- (0,3.2) node[above] {$y$};
  % 电荷

  \fill (0,{\a}) circle (2pt) node[left] {$(0,a)$};
  \fill ({\a},{\a}) circle (2pt) node[above right] {$P(a,a)$};
  \fill (0,{2*\a}) circle (2pt) node[above left] {$Q$};
  % 电荷符号标注

  \node[red] at (-0.35,{\a+0.3}) {$-q$};
  % 距离示意虚线
  \draw[dashed] (0,{\a}) -- ({\a},{\a});
  \draw[dashed] ({\a},0) -- ({\a},{\a});
  \draw[dashed] ({\a},{\a}) -- (0,{2*\a});
  \node[below left] at (0.42,0.25) {$O$};
