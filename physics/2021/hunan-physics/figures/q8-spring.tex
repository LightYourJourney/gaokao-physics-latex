% 第 8 题图甲：A、B 两物体经轻弹簧相连，B 靠墙面，外力 F 作用在 A 上
\usetikzlibrary{decorations.pathmorphing}
\begin{tikzpicture}[font=\small, >=Stealth, scale=1.0]
  % 地面
  \draw[thick] (-3.4,0) -- (3.4,0);
  \foreach \x in {-3.2,-2.9,...,3.2} \draw (\x,0) -- (\x-0.08,-0.12);
  % B 靠墙（右侧墙）
  \draw[thick, fill=gray!20] (2.8,0) rectangle (3.1,1.2);
  % A 物块
  \draw[fill=white, thick] (-2.2,0.1) rectangle (-1.2,1.0);
  \node at (-1.7,0.55) {$A$};
  % B 物块
  \draw[fill=white, thick] (1.6,0.1) rectangle (2.6,1.0);
  \node at (2.1,0.55) {$B$};
  % 弹簧（A 与 B 之间）
  \draw[thick, decorate, decoration={coil, segment length=3.2pt, amplitude=3.5pt}]
    (-1.2,0.55) -- (1.6,0.55);
  % 外力 F 作用在 A 上（水平向左）
  \draw[->, very thick] (-0.7,0.55) -- (0.3,0.55);
  \node[above] at (-0.2,0.62) {$F$};
\end{tikzpicture}
