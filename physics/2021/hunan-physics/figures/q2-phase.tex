% 第 2 题：p-x 相轨迹四选项图（1×4 小坐标系）
% 使用 groupplot 固定四图的行列关系，坐标范围与曲线拓扑按原题图逐项设置。
\begin{tikzpicture}[font=\small]
  \begin{groupplot}[
      group style={group size=4 by 1, horizontal sep=0.55cm},
      width=3.2cm, height=2.6cm,
      axis lines=middle, clip=false,
      xmin=-0.2, xmax=1.25, ymin=-0.6, ymax=1.25,
      xtick=\empty, ytick=\empty,
      xlabel={$x$}, ylabel={$p$},
      xlabel style={anchor=west}, ylabel style={anchor=south}]
    \nextgroupplot[xmin=-1.2, xmax=1.2]
      \addplot[thick, domain=-1:1, samples=80] {x^2};
      \node at (axis description cs:0.5,-0.14) {A};
    \nextgroupplot
      \addplot[thick, domain=0:1, samples=80] {x^2};
      \node at (axis description cs:0.5,-0.14) {B};
    \nextgroupplot
      \addplot[thick, domain=0:1, samples=80] {sqrt(x)};
      \addplot[thick, domain=0:1, samples=80] {-sqrt(x)};
      \node at (axis description cs:0.5,-0.14) {C};
    \nextgroupplot
      \addplot[thick, domain=0:1.2, samples=80] {sqrt(x)};
      \node at (axis description cs:0.5,-0.14) {D};
  \end{groupplot}
\end{tikzpicture}
