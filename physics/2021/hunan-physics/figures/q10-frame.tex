% 第 10 题：两个正方形金属框经绝缘轻质杆相连的组合体，下方为匀强磁场区域
% 框边长 l、杆长 l；组合体下底边距磁场区域上边界 h；磁场区域高 l、水平足够宽。
\begin{tikzpicture}[font=\small, >=Stealth, scale=0.9]
  \def\L{1.5}
  % 坐标系（x 向右、y 向上，原点取磁场下边界中心）
  \draw[->] (-2.8,0) -- (3.4,0) node[right] {$x$};
  \draw[->] (0,-0.4) -- (0,5.2) node[above] {$y$};
  % 磁场区域（高 l，水平条带）
  \draw[thick] (-2.6,0) rectangle (2.6,{\L});
  \foreach \xx in {-2.1,-1.4,...,2.1} {
    \foreach \yy in {0.4,1.0} {
      \node at (\xx,\yy) {$\times$};
    }
  }
  \node at (2.9,0.75) {$\gv{B}$};
  % 组合体：下框、上框与绝缘杆
  \draw[thick] (-0.75,{0.75+\L+0.0}) -- (-0.75,{2.0}) -- (0.75,2.0) -- (0.75,{0.75}); % 下框
  \draw[thick] (-0.75,{0.75}) rectangle (0.75,{2.0});
  \draw[thick] (-0.75,{2.0+1.5}) rectangle (0.75,{2.0+1.5+1.25}); % 上框
  % 绝缘杆（下框上边到上框下边，长 l）
  \draw[thick] (-0.3,{2.0}) -- (-0.3,{2.0+1.5});
  \draw[thick] (0.3,{2.0}) -- (0.3,{2.0+1.5});
  % 初速度箭头
  \draw[->, very thick] (-1.8,{2.0+0.9}) -- (-0.8,{2.0+0.9});
  \node[above] at (-1.3,{2.0+0.95}) {$v_{0}$};
  % 高度标注 h、l
  \draw[<->] (-2.4,{1.5+1.25}) -- (-2.4,{1.5+1.25+1.5}) node[midway, left] {$l$};
  \draw[<->] (-2.4,{1.5+1.25+1.5}) -- (-2.4,{3.25+1.5}) node[midway, left] {$h$};
  \draw[<->] (-2.4,0) -- (-2.4,1.5) node[midway, left] {$l$};
  \draw[dashed] (-2.6,{1.5}) -- (2.6,{1.5});
\end{tikzpicture}
