% 第 2 题：物块沿光滑斜面上滑的四个候选图像
% 使用 groupplot 固定四图的行列关系；旧版使用绝对 at 坐标，导致 D 图进入第二行并与题干错位。
\begin{tikzpicture}[font=\small]
  \begin{groupplot}[
      group style={group size=4 by 1, horizontal sep=0.42cm},
      width=3.25cm, height=3.05cm,
      axis lines=middle,
      xmin=0, xmax=4.4, ymin=0, ymax=4.6,
      xtick=\empty, ytick=\empty,
      xlabel style={anchor=west}, ylabel style={anchor=south}]
    \nextgroupplot[xlabel={$x$}, ylabel={$v_x$}]
      \addplot[thick] coordinates {(0,4.2) (3.3,0)};
      \node[below left] at (0.55,4.1) {A};
    \nextgroupplot[xlabel={$x$}, ylabel={$v_x$}]
      \addplot[thick] coordinates {(0,0) (4,3.8)};
      \node[below left] at (0.55,0.25) {B};
    \nextgroupplot[xlabel={$y$}, ylabel={$v_y$}]
      \addplot[thick, domain=0:4, samples=60] {sqrt(9-2.25*x)};
      \node[below left] at (0.55,4.1) {C};
    \nextgroupplot[xlabel={$y$}, ylabel={$v_y$}]
      \addplot[thick] coordinates {(0,3.1) (4,0)};
      \node[below left] at (0.55,3.0) {D};
  \end{groupplot}
\end{tikzpicture}
