% 第 7 题：两波源与 C 处质点的坐标平面（TikZ）
% A(0,0)、B(4,0)、C(0,3)；波沿 AC、BC 方向传播。
\begin{tikzpicture}[font=\small, scale=1.0]
  % 网格点（单位间距）
  \foreach \x in {0,...,4} {
    \foreach \y in {0,...,3} {
      \fill[gray!50] (\x,\y) circle (0.04);
    }
  }
  % 坐标轴
  \draw[-{Stealth}] (-0.4,0) -- (4.6,0) node[right] {$x$};
  \draw[-{Stealth}] (0,-0.4) -- (0,3.6) node[above] {$y$};
  % 波传播方向：沿 AC（竖直向上）、BC（指向 C）
  \draw[-{Stealth}, thick] (0,0.8) -- (0,2.2);
  \draw[-{Stealth}, thick] (4,0.6) -- (1.1,2.4);
  % 点 A、B、C
  \fill (0,0) circle (0.07);
  \fill (4,0) circle (0.07);
  \fill (0,3) circle (0.07);
  \node[below left] at (0,0) {$A$};
  \node[below right] at (4,0) {$B$};
  \node[above left] at (0,3) {$C$};
\end{tikzpicture}
