% 第 14 题：圆筒与磁场/电场区域（TikZ 线稿，侧视剖面）
% 筒内 x≤0 为匀强磁场（沿 +x）；筒外 x≥0 为匀强电场（沿 +y）；
% 左右端面圆心 O'、O 处各开小孔，电子枪在 O' 处向筒内发射电子
\begin{tikzpicture}[font=\small, >=Stealth]
  % 坐标轴（原点 O 在右端面圆心）
  \draw[->] (-5.8,0) -- (3.6,0) node[below] {$x$};
  \draw[->] (0,-1.9) -- (0,2.7) node[left] {$y$};
  % 圆筒（x ∈ [-L, 0]）
  \draw[thick] (-4.6,1.2) -- (0,1.2);
  \draw[thick] (-4.6,-1.2) -- (0,-1.2);
  \draw[very thick] (-4.6,1.35) -- (-4.6,-1.35);
  \draw[very thick] (0,1.35) -- (0,-1.35);
  % 小孔 O' 与 O
  \fill (-4.6,0) circle (1.5pt) node[above left] {$O'$};
  \fill (0,0) circle (1.5pt) node[below right] {$O$};
  % 尺寸标注：内半径 2r、长度 L
  \draw[<->] (-5.0,0) -- (-5.0,1.2) node[midway, left] {$2r$};
  \draw[<->] (-4.6,-1.75) -- (0,-1.75) node[midway, below] {$L$};
  % 电子枪（左端外侧）
  \draw[thick] (-6.0,0.5) rectangle (-5.1,-0.5);
  \node[below] at (-5.55,-0.85) {电子枪};
  \node[above] at (-5.55,0.85) {向筒内多个方向};
  % 筒内磁场（×，沿 +x）
  \foreach \x in {-4.1,-3.3,-2.5,-1.7,-0.9} {
    \foreach \y in {-0.7,0,0.7} {
      \node[gray] at (\x,\y) {$\times$};
    }
  }
  \node[gray] at (-2.3,1.55) {$\vec{B}$ 沿 $+x$};
  % 筒外电场（+y 方向箭头）
  \foreach \x in {1.0,1.8,2.6} {
    \draw[->, gray] (\x,-1.35) -- (\x,1.35);
  }
  \node[gray] at (1.8,1.9) {$\vec{E}$ 沿 $+y$};
  % 说明
  \node at (0.6,-2.35) {圆筒内半径 $2r$、长 $L$，筒壁不计};
\end{tikzpicture}
