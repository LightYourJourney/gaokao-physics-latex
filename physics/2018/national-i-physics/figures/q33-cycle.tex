\begin{tikzpicture}[font=\small, >=Stealth, scale=1.0]
  \draw[->] (0,0) -- (5.4,0) node[right] {$V$};
  \draw[->] (0,0) -- (0,4.2) node[above] {$T$};
  \node[below left] at (0,0) {$O$};
  \coordinate (a) at (1.3,1.0);
  \coordinate (b) at (1.3,2.2);
  \coordinate (c) at (2.9,3.3);
  \coordinate (d) at (4.5,3.3);
  \coordinate (e) at (4.5,1.0);
  % 过程①：a→b（等容升压/升温）
  \draw[->,thick] (a) -- (b);
  % 过程②：b→c
  \draw[->,thick] (b) -- (c);
  % 过程③：c→d（等温膨胀）
  \draw[->,thick] (c) -- (d);
  % 过程④：d→e（等容降温）
  \draw[->,thick] (d) -- (e);
  % 等温虚线
  \draw[dashed] (0,3.3) -- (c);

  \draw[dashed] (e) -- (4.5,0);
  % 点标注
  \foreach \p/\lab/\pos in {a/a/below left,b/b/below right,c/c/above,d/d/above,e/e/right} {

    \node[\pos] at (\p) {$\lab$};
  }
  \node at (0.75,1.65) {①};
  \node at (2.0,2.5) {②};
