% 第 6 题：三个平行金属板 A/B/C（TikZ 线稿）
% A、C 接地；B 板到 A、C 距离分别为 d1、d2（d1<d2）；a、b 为场中两点
\begin{tikzpicture}[font=\small, >=Stealth]
  % 板 A、B、C（竖线表示板面）
  \draw[thick] (0,-1.2) -- (0,1.2) node[above]{$A$};
  \draw[thick] (2.2,-1.2) -- (2.2,1.2) node[above]{$B$};
  \draw[thick] (4.4,-1.2) -- (4.4,1.2) node[above]{$C$};

  \draw (0,-1.2) -- (0,-1.7); \draw (-0.3,-1.9) -- (0.3,-1.9);

  \draw (4.2,-2.1) -- (4.6,-2.1); \draw (4.3,-2.3) -- (4.5,-2.3);
  % 距离标注
  \draw[<->] (0,1.35) -- (2.2,1.35) node[midway, above]{$d_1$};
  \draw[<->] (2.2,1.35) -- (4.4,1.35) node[midway, above]{$d_2$};
